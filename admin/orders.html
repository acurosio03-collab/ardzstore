// ========================= // ARDZ STORE - ORDERS.JS // ========================= let orders = JSON.parse(localStorage.getItem("orders")) || []; function rupiah(angka){ return "Rp " + Number(angka).toLocaleString("id-ID"); } // ========================= // TAMPILKAN PESANAN // ========================= function tampilkanOrders(data = orders){ const list = document.getElementById("ordersList"); if(!list) return; list.innerHTML = ""; let menunggu = 0; let diproses = 0; let selesai = 0; data.forEach((item,index)=>{ let statusClass = ""; if(item.status=="Menunggu"){ menunggu++; statusClass="status-menunggu"; } if(item.status=="Diproses"){ diproses++; statusClass="status-proses"; } if(item.status=="Selesai"){ selesai++; statusClass="status-selesai"; } list.innerHTML += `

🎮 ${item.game}

User ID : ${item.userid}

${item.zoneid ? `

Zone ID : ${item.zoneid}

` : ""}

Produk : ${item.produk}

Total : ${rupiah(item.total)}

Pembayaran : ${item.payment}

WhatsApp : ${item.wa}

Waktu : ${item.waktu}

Status : ${item.status}

`; }); document.getElementById("totalOrder").innerHTML = orders.length; document.getElementById("waitingOrder").innerHTML = menunggu; document.getElementById("processOrder").innerHTML = diproses; document.getElementById("doneOrder").innerHTML = selesai; if(data.length==0){ list.innerHTML="

Tidak ada pesanan.

"; } } // ========================= // UBAH STATUS // ========================= function ubahStatus(index){ if(orders[index].status=="Menunggu"){ orders[index].status="Diproses"; }else if(orders[index].status=="Diproses"){ orders[index].status="Selesai"; }else{ orders[index].status="Menunggu"; } localStorage.setItem("orders",JSON.stringify(orders)); tampilkanOrders(); } // ========================= // HAPUS // ========================= function hapusOrder(index){ if(confirm("Hapus pesanan ini?")){ orders.splice(index,1); localStorage.setItem("orders",JSON.stringify(orders)); tampilkanOrders(); } } // ========================= // CARI PESANAN // ========================= function cariOrder(){ let keyword = document .getElementById("searchOrder") .value .toLowerCase(); let hasil = orders.filter(item=>{ return ( item.game.toLowerCase().includes(keyword) || item.userid.toLowerCase().includes(keyword) ); }); tampilkanOrders(hasil); } // ========================= tampilkanOrders();
